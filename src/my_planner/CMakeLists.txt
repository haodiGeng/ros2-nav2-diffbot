cmake_minimum_required(VERSION 3.8)
project(my_planner)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rclcpp_lifecycle rcl_interfaces nav2_core nav2_costmap_2d nav_msgs geometry_msgs pluginlib tf2_ros std_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_library(my_astar_core STATIC src/astar_core.cpp)
set_target_properties(my_astar_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(my_astar_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
add_library(my_astar_planner SHARED src/astar_planner.cpp)
target_link_libraries(my_astar_planner my_astar_core)
ament_target_dependencies(my_astar_planner rclcpp rclcpp_lifecycle rcl_interfaces nav2_core nav2_costmap_2d nav_msgs geometry_msgs pluginlib tf2_ros std_msgs)
pluginlib_export_plugin_description_file(nav2_core astar_planner_plugin.xml)
install(TARGETS my_astar_core my_astar_planner ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_astar_core test/test_astar_core.cpp)
  target_link_libraries(test_astar_core my_astar_core)
endif()
ament_export_include_directories(include)
ament_export_libraries(my_astar_core my_astar_planner)
ament_export_dependencies(nav2_core nav2_costmap_2d pluginlib)
ament_package()
